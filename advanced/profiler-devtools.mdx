## Overview

RyunixJS includes a powerful performance profiler and development tools for measuring render times, identifying bottlenecks, and optimizing your application.

## Profiler API

The global profiler instance tracks component render performance automatically.

### Basic Usage

```javascript
import { profiler } from 'ryunix';

// Get performance statistics
const stats = profiler.getStats();
console.log(stats);
// {
//   total: 1234.56,
//   avg: 12.34,
//   max: 45.67,
//   min: 2.34,
//   count: 100
// }

// Log stats to console
profiler.logStats();
```

#### profiler.recordRender(componentName, duration)

Manually record component render time:

```javascript

```

**Source:** `/home/daytona/workspace/source/packages/core/src/lib/profiler.js:32-44`

**Implementation:**

```javascript
recordRender(componentName, duration) {
  if (!this.enabled) return;
  
  this.renderTimes.push({
    component: componentName,
    duration,
    timestamp: Date.now(),
  });
  
  if (this.renderTimes.length > this.maxSamples) {
    this.renderTimes.shift();
  }
}
```

#### profiler.getStats()

Get aggregate performance statistics:

```javascript
const stats = profiler.getStats();
console.log(`Average render: ${stats.avg.toFixed(2)}ms`);
console.log(`Max render: ${stats.max.toFixed(2)}ms`);
```

**Returns:**
- `total` - Total render time
- `avg` - Average render time
- `max` - Longest render time
- `min` - Shortest render time
- `count` - Number of renders tracked

**Source:** `/home/daytona/workspace/source/packages/core/src/lib/profiler.js:46-55`

#### profiler.getSlowestComponents(limit)

Identify performance bottlenecks:

```javascript
const slowest = profiler.getSlowestComponents(5);
slowest.forEach(comp => {
  console.log(`${comp.name}: ${comp.avg.toFixed(2)}ms avg`);
});
```

**Parameters:**
- `limit` (number) - Number of components to return (default: 10)

**Returns:** Array of components sorted by average render time

**Source:** `/home/daytona/workspace/source/packages/core/src/lib/profiler.js:57-81`

**Implementation:**

```javascript
getSlowestComponents(limit = 10) {
  if (!this.enabled) return [];
  
  const byComponent = new Map();
  
  this.renderTimes.forEach(({ component, duration }) => {
    if (!byComponent.has(component)) {
      byComponent.set(component, { total: 0, count: 0, max: 0 });
    }
    const stats = byComponent.get(component);
    stats.total += duration;
    stats.count++;
    stats.max = Math.max(stats.max, duration);
  });
  
  return Array.from(byComponent.entries())
    .map(([name, stats]) => ({
      name,
      avg: stats.total / stats.count,
      max: stats.max,
      count: stats.count,
    }))
    .sort((a, b) => b.avg - a.avg)
    .slice(0, limit);
}
```

## useProfiler Hook

Profile individual component renders:

```jsx
import { useProfiler } from 'ryunix';

function ExpensiveComponent() {
  const endProfile = useProfiler('ExpensiveComponent');
  
  // Component logic here
  
  endProfile(); // Call when render completes
  
  return <div>Content</div>;
}
```

### API Signature

```javascript
const useProfiler = (componentName) => Function
```

**Parameters:**
- `componentName` (string) - Name to identify in profiler

**Returns:**
- Function to call when profiling should end

**Source:** `/home/daytona/workspace/source/packages/core/src/lib/profiler.js:128-135`

**Implementation:**

```javascript
const useProfiler = (componentName) => {
  const startTime = perfNow();
  
  return () => {
    const duration = perfNow() - startTime;
    profiler.recordRender(componentName, duration);
  };
}
```

## withProfiler HOC

Automatically profile components with a Higher-Order Component:

```jsx
import { withProfiler } from 'ryunix';

const DataTable = ({ data }) => {
  return (
    <table>
      {data.map(row => <tr key={row.id}>...</tr>)}
    </table>
  );
};

// Wrap component to enable profiling
const ProfiledDataTable = withProfiler(DataTable, 'DataTable');

export default ProfiledDataTable;
```

### API Signature

```javascript
const withProfiler = (Component, name) => ProfiledComponent
```

**Parameters:**
- `Component` (Function) - Component to profile
- `name` (string) - Name for profiler tracking

**Returns:**
- Wrapped component that records render times

**Source:** `/home/daytona/workspace/source/packages/core/src/lib/profiler.js:140-148`

**Implementation:**

```javascript
const withProfiler = (Component, name) => {
  return (props) => {
    profiler.startMeasure(name);
    const result = Component(props);
    const duration = profiler.endMeasure(name);
    if (duration) profiler.recordRender(name, duration);
    return result;
  };
}
```

## Practical Examples

### Monitoring Dashboard

Create a real-time performance dashboard:

```jsx
import { profiler, useStore, useEffect } from 'ryunix';

function PerformanceDashboard() {
  const [stats, setStats] = useStore(null);
  const [slowest, setSlowest] = useStore([]);
  
  useEffect(() => {
    const interval = setInterval(() => {
      setStats(profiler.getStats());
      setSlowest(profiler.getSlowestComponents(10));
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  if (!stats) return <div>Loading...</div>;
  
  return (
    <div className="dashboard">
      <h2>Performance Stats</h2>
      <div className="stats-grid">
        <StatCard 
          label="Avg Render" 
          value={`${stats.avg.toFixed(2)}ms`} 
        />
        <StatCard 
          label="Max Render" 
          value={`${stats.max.toFixed(2)}ms`} 
        />
        <StatCard 
          label="Total Renders" 
          value={stats.count} 
        />
      </div>
      
      <h3>Slowest Components</h3>
      <table>
        <thead>
          <tr>
            <th>Component</th>
            <th>Avg Time</th>
            <th>Max Time</th>
            <th>Renders</th>
          </tr>
        </thead>
        <tbody>
          {slowest.map((comp, i) => (
            <tr key={i}>
              <td>{comp.name}</td>
              <td>{comp.avg.toFixed(2)}ms</td>
              <td>{comp.max.toFixed(2)}ms</td>
              <td>{comp.count}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```

### Performance Budget

Warn when components exceed performance budgets:

```jsx
import { withProfiler, profiler } from 'ryunix';

function withPerformanceBudget(Component, name, budgetMs) {
  return withProfiler((props) => {
    const result = Component(props);
    
    // Check if last render exceeded budget
    const stats = profiler.getStats();
    if (stats && stats.max > budgetMs) {
      console.warn(
        `⚠️  ${name} exceeded performance budget: ` +
        `${stats.max.toFixed(2)}ms > ${budgetMs}ms`
      );
    }
    
    return result;
  }, name);
}

// Usage
const DataTable = withPerformanceBudget(
  DataTableComponent, 
  'DataTable', 
  50 // 50ms budget
);
```

## Best Practices

### 1. Profile in Development

Run profiler during development to catch issues early:

```javascript
if (process.env.NODE_ENV === 'development') {
  setInterval(() => {
    profiler.logStats();
  }, 10000); // Log every 10 seconds
}
```

### 2. Focus on Slowest Components

Prioritize optimization by focusing on top offenders:

```javascript
const slowest = profiler.getSlowestComponents(3);
if (slowest[0]?.avg > 50) {
  console.warn('Optimization needed:', slowest[0].name);
}
```

### 3. Measure Before Optimizing

Always profile before and after optimization:

```javascript
profiler.clear();
// Run component
const before = profiler.getStats();

// Apply optimization

profiler.clear();
// Run component
const after = profiler.getStats();

console.log(
  `Improvement: ${before.avg - after.avg}ms`
);
```

### 4. Use with Memo

Combine profiling with memoization:

```jsx
import { memo, withProfiler } from 'ryunix';

const OptimizedComponent = memo(
  withProfiler(Component, 'Component')
);
```

## Performance Tips

- Profiler has minimal overhead in development

- Uses efficient Map/Array data structures
- Automatically limits stored samples
